# ============================================================
# EXPERIMENT 1 — LRU vs MRU
# ============================================================
# Buffer pool size = 16, index = bplus_tree
# Workload: sequential (1000 records, 200 queries) and random (1000 records, 200 queries)
# Seed: 42

# Sequential + LRU
rm -f *.dat *.idx catalog.dat output.txt log.csv stats_output.txt
python3 workload_generator.py --mode sequential --records 1000 --queries 200 --seed 42 --stats-reset > /tmp/exp1_seq.txt
python3 -c "
import json
c = json.load(open('config.json'))
c.update({'buffer_pool_size':16,'replacement_policy':'LRU','index_strategy':'bplus_tree'})
json.dump(c, open('/tmp/exp1_cfg.json','w'))
"
python3 archive.py /tmp/exp1_cfg.json /tmp/exp1_seq.txt
echo "=== Sequential + LRU ===" && cat stats_output.txt

# ============================================================
# EXPERIMENT 3 — Buffer Pool Size Sensitivity
# ============================================================
# Policy = LRU, index = bplus_tree
# Workload: random (1000 records, 200 queries)
# Seed: 42

python3 workload_generator.py --mode random --records 1000 --queries 200 --seed 42 --stats-reset > /tmp/exp3_wl.txt

for SIZE in 4 8 16 32 64; do
  rm -f *.dat *.idx catalog.dat output.txt log.csv stats_output.txt
  python3 -c "
import json
c = json.load(open('config.json'))
c.update({'buffer_pool_size':${SIZE},'replacement_policy':'LRU','index_strategy':'bplus_tree'})
json.dump(c, open('/tmp/exp3_cfg.json','w'))
"
  python3 archive.py /tmp/exp3_cfg.json /tmp/exp3_wl.txt
  echo "=== pool_size=${SIZE} ===" && cat stats_output.txt
done
